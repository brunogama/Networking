```swift
import XCTest
import SwiftSyntax
@testable import MacroTemplateKit

final class RendererTests: XCTestCase {

  // MARK: - Literal Rendering

  func testRenderLiteral_integer() {
    let template: Template<Void> = .literal(.integer(42))
    let result = Renderer.render(template)
    XCTAssertTrue(result.is(IntegerLiteralExprSyntax.self))
    // Optionally check: result.description.contains("42")
  }

  func testRenderLiteral_string() {
    let template: Template<Void> = .literal(.string("hello"))
    let result = Renderer.render(template)
    XCTAssertTrue(result.is(StringLiteralExprSyntax.self))
  }

  func testRenderLiteral_boolean() {
    let templateTrue: Template<Void> = .literal(.boolean(true))
    let templateFalse: Template<Void> = .literal(.boolean(false))
    XCTAssertTrue(Renderer.render(templateTrue).is(BooleanLiteralExprSyntax.self))
    XCTAssertTrue(Renderer.render(templateFalse).is(BooleanLiteralExprSyntax.self))
  }

  func testRenderLiteral_nil() {
    let template: Template<Void> = .literal(.nil)
    let result = Renderer.render(template)
    XCTAssertTrue(result.is(NilLiteralExprSyntax.self))
  }

  // MARK: - Variable Rendering

  func testRenderVariable() {
    let template: Template<String> = .variable("myVar", payload: "metadata")
    let result = Renderer.render(template)
    // Should produce identifier expression
    XCTAssertTrue(result.description.contains("myVar"))
  }

  // MARK: - Compound Expression Rendering

  func testRenderConditional() {
    let template: Template<Void> = .conditional(
      condition: .literal(.boolean(true)),
      thenBranch: .literal(.integer(1)),
      elseBranch: .literal(.integer(0))
    )
    let result = Renderer.render(template)
    XCTAssertTrue(result.is(TernaryExprSyntax.self))
  }

  func testRenderFunctionCall() {
    let template: Template<Void> = .functionCall(
      function: .variable("print", payload: ()),
      arguments: [.literal(.string("hello"))]
    )
    let result = Renderer.render(template)
    XCTAssertTrue(result.is(FunctionCallExprSyntax.self))
  }

  func testRenderBinaryOperation() {
    let template: Template<Void> = .binaryOperation(
      left: .literal(.integer(1)),
      operator: "+",
      right: .literal(.integer(2))
    )
    let result = Renderer.render(template)
    XCTAssertTrue(result.is(InfixOperatorExprSyntax.self))
  }

  func testRenderPropertyAccess() {
    let template: Template<Void> = .propertyAccess(
      base: .variable("object", payload: ()),
      property: "property"
    )
    let result = Renderer.render(template)
    XCTAssertTrue(result.is(MemberAccessExprSyntax.self))
  }

  func testRenderArrayLiteral() {
    let template: Template<Void> = .arrayLiteral([
      .literal(.integer(1)),
      .literal(.integer(2)),
      .literal(.integer(3))
    ])
    let result = Renderer.render(template)
    XCTAssertTrue(result.is(ArrayExprSyntax.self))
  }

  // MARK: - Edge Cases

  func testRenderNestedExpressions() {
    // property.method(arg1, arg2 + arg3)
    let template: Template<Void> = .functionCall(
      function: .propertyAccess(
        base: .variable("property", payload: ()),
        property: "method"
      ),
      arguments: [
        .variable("arg1", payload: ()),
        .binaryOperation(
          left: .variable("arg2", payload: ()),
          operator: "+",
          right: .variable("arg3", payload: ())
        )
      ]
    )
    let result = Renderer.render(template)
    XCTAssertTrue(result.is(FunctionCallExprSyntax.self))
  }
}
```

Use `ExprSyntax.is(_:)` for type checking. Use `.description` for string content verification.
