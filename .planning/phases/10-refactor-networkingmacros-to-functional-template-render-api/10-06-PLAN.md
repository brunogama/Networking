import MacroTemplateKit
import SwiftSyntax
import SwiftSyntaxBuilder

/// Shared template helpers for HTTP macro implementations.
public enum HTTPMacroTemplate {

  // MARK: - Request Building

  /// Creates a URLRequest template with method and URL.
  public static func urlRequest<Method: HTTPMethodWithBodyConstraint>(
    method: Method.Type,
    url: Template<Void>,
    headers: [(name: String, value: Template<Void>)] = [],
    body: Template<Void>? = nil
  ) -> Template<Void> {
    var components: [Template<Void>] = [
      // let request = URLRequest(url: url)
      .variableDeclaration(
        name: "request",
        type: "URLRequest",
        initializer: .function("URLRequest", arguments: [
          (label: "url", value: url)
        ])
      ),

      // request.httpMethod = "METHOD"
      .binaryOperation(
        left: .propertyAccess(
          base: .variable("request", payload: ()),
          property: "httpMethod"
        ),
        operator: "=",
        right: .literal(Method.methodName)
      )
    ]

    // Add headers
    for header in headers {
      components.append(
        .function("setValue", arguments: [
          (label: nil, value: header.value),
          (label: "forHTTPHeaderField", value: .literal(header.name))
        ])
      )
    }

    // Add body if provided
    if let body = body {
      components.append(
        .binaryOperation(
          left: .propertyAccess(
            base: .variable("request", payload: ()),
            property: "httpBody"
          ),
          operator: "=",
          right: body
        )
      )
    }

    return .arrayLiteral(components)
  }

  // MARK: - Path Template Expansion

  /// Creates URL template from path template string.
  ///
  /// Handles path parameters like "/users/{id}" → baseURL.appendingPathComponent("users").appendingPathComponent(id)
  public static func urlFromPathTemplate(
    _ pathTemplate: String,
    baseURL: Template<Void>,
    pathParameters: [String: Template<Void>]
  ) -> Template<Void> {
    let segments = parsePathSegments(pathTemplate)

    var result = baseURL
    for segment in segments {
      if segment.hasPrefix("{") && segment.hasSuffix("}") {
        // Path parameter
        let paramName = String(segment.dropFirst().dropLast())
        if let paramValue = pathParameters[paramName] {
          result = .function("appendingPathComponent", arguments: [
            (label: nil, value: result),
            (label: nil, value: paramValue)
          ])
        }
      } else {
        // Static segment
        result = .function("appendingPathComponent", arguments: [
          (label: nil, value: result),
          (label: nil, value: .literal(segment))
        ])
      }
    }

    return result
  }

  // MARK: - Response Handling

  /// Creates response decoding template.
  public static func decodeResponse(
    type: String,
    from data: Template<Void>,
    decoder: Template<Void> = .variable("decoder", payload: ())
  ) -> Template<Void> {
    .function("decode", arguments: [
      (label: nil, value: .variable(type, payload: ())),
      (label: "from", value: data)
    ])
  }

  // MARK: - Private Helpers

  private static func parsePathSegments(_ path: String) -> [String] {
    path.split(separator: "/")
      .map(String.init)
      .filter { !$0.isEmpty }
  }
}

// MARK: - Syntax Factory Extension

extension HTTPMacroTemplate {
  /// Renders a template to ExprSyntax.
  public static func render(_ template: Template<Void>) -> ExprSyntax {
    Renderer.render(template)
  }

  /// Renders a typed template to ExprSyntax.
  public static func render<M: HTTPMethodWithBodyConstraint>(
    _ template: TypedTemplate<M>
  ) -> ExprSyntax {
    template.render()
  }
}
