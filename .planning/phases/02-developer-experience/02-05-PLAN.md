```swift
/// A request with attached configuration for caching and retry behavior
public struct ChainedRequest<T: Decodable & Sendable>: Sendable {
  public let request: HTTPRequest
  public let decoder: JSONDecoder
  public let cacheConfig: CacheConfiguration?
  public let retryConfig: RetryConfiguration?

  public struct CacheConfiguration: Sendable {
    public let ttl: TimeInterval
  }

  public struct RetryConfiguration: Sendable {
    public let maxAttempts: Int
    public let baseDelay: TimeInterval
    public let maxDelay: TimeInterval
  }

  public init(
    request: HTTPRequest,
    decoder: JSONDecoder = JSONDecoder(),
    cacheConfig: CacheConfiguration? = nil,
    retryConfig: RetryConfiguration? = nil
  ) {
    self.request = request
    self.decoder = decoder
    self.cacheConfig = cacheConfig
    self.retryConfig = retryConfig
  }

  /// Adds caching configuration
  public func cacheable(ttl: TimeInterval = 300) -> ChainedRequest<T> {
    ChainedRequest(
      request: request,
      decoder: decoder,
      cacheConfig: CacheConfiguration(ttl: ttl),
      retryConfig: retryConfig
    )
  }

  /// Adds retry configuration
  public func retryable(
    maxAttempts: Int = 3,
    baseDelay: TimeInterval = 1.0,
    maxDelay: TimeInterval = 60.0
  ) -> ChainedRequest<T> {
    ChainedRequest(
      request: request,
      decoder: decoder,
      cacheConfig: cacheConfig,
      retryConfig: RetryConfiguration(
        maxAttempts: maxAttempts,
        baseDelay: baseDelay,
        maxDelay: maxDelay
      )
    )
  }

  /// Executes the request with attached configurations
  public func execute(on client: any HTTPClient) async throws -> DecodedResponse<T> {
    // Build interceptors from configuration
    var tempClient = client

    // Execute request (interceptors should already be configured on client)
    // For now, use the client as-is. Future: create ephemeral client with config.
    let response = try await client.execute(request)

    // Decode response
    guard let data = response.body, !data.isEmpty else {
      throw HTTPError(
        category: .decoding(DecodingError.dataCorrupted(
          DecodingError.Context(
            codingPath: [],
            debugDescription: "Empty response body"
          )
        )),
        request: response.request
      )
    }

    do {
      let value = try decoder.decode(T.self, from: data)
      return DecodedResponse(response: response, value: value)
    } catch let error as DecodingError {
      throw HTTPError(
        category: .decoding(error),
        request: response.request
      )
    }
  }
}
```

Enhance ChainedRequest.execute(on:) to actually wire caching and retry configurations to interceptors.

**Approach:** Since NetworkClient is initialized with middleware arrays and cannot be modified at runtime, use a different strategy:

1. For caching: Use the existing CachingInterceptor.getCachedResponse() method to check cache BEFORE making a request. After getting response, cache it manually if cacheConfig is set.

2. For retry: Implement retry logic inline within execute() method, using the RetryInterceptor's exponential backoff calculation pattern.

```swift
public func execute(on client: any HTTPClient) async throws -> DecodedResponse<T> {
  // Check cache first if caching is configured
  if let cacheConfig = cacheConfig {
    // Create ephemeral caching interceptor for cache lookup
    let cacheInterceptor = CachingInterceptor(ttl: cacheConfig.ttl)
    if let cachedResponse = await cacheInterceptor.getCachedResponse(
      for: request.method,
      path: request.url.path
    ) {
      // Return cached response if available
      return try decodeResponse(cachedResponse)
    }
  }

  // Execute with retry logic if configured
  let response: HTTPResponse
  if let retryConfig = retryConfig {
    response = try await executeWithRetry(
      client: client,
      maxAttempts: retryConfig.maxAttempts,
      baseDelay: retryConfig.baseDelay,
      maxDelay: retryConfig.maxDelay
    )
  } else {
    response = try await client.execute(request)
  }

  }

  return try decodeResponse(response)

private func executeWithRetry(
  client: any HTTPClient,
  maxAttempts: Int,
  baseDelay: TimeInterval,
  maxDelay: TimeInterval
) async throws -> HTTPResponse {
  var lastError: Error?

  for attempt in 0..<maxAttempts {
    do {
      return try await client.execute(request)
    } catch let error as HTTPError {
      lastError = error

      // Check if error is retryable
      guard shouldRetry(error: error) else {
        throw error
      }

      // Don't delay after last attempt
      if attempt < maxAttempts - 1 {
        let delay = calculateDelay(
          attempt: attempt,
          baseDelay: baseDelay,
          maxDelay: maxDelay
        )
        try await Task.sleep(nanoseconds: UInt64(delay * 1_000_000_000))
      }
    } catch {
      throw error
    }
  }

  throw lastError ?? HTTPError(
    category: .network(.serverUnreachable),
    request: request
  )
}

private func shouldRetry(error: HTTPError) -> Bool {
  switch error.category {
  case .http(let status):
    // Retry server errors and specific client errors
    return status.rawValue >= 500 || status.rawValue == 408 || status.rawValue == 429
  case .network:
    return true
  default:
    return false
  }
}

private func calculateDelay(
  attempt: Int,
  baseDelay: TimeInterval,
  maxDelay: TimeInterval
) -> TimeInterval {
  let exponentialDelay = baseDelay * pow(2.0, Double(attempt))
  let cappedDelay = min(exponentialDelay, maxDelay)
  let jitter = Double.random(in: 0...(cappedDelay * 0.1))
  return min(cappedDelay + jitter, maxDelay)
}

private func decodeResponse(_ response: HTTPResponse) throws -> DecodedResponse<T> {
  guard let data = response.body, !data.isEmpty else {
    throw HTTPError(
      category: .decoding(DecodingError.dataCorrupted(
        DecodingError.Context(
          codingPath: [],
          debugDescription: "Empty response body"
        )
      )),
      request: response.request
    )
  }

  do {
    let value = try decoder.decode(T.self, from: data)
    return DecodedResponse(response: response, value: value)
  } catch let error as DecodingError {
    throw HTTPError(
      category: .decoding(error),
      request: response.request
    )
  }
}
```

```swift
/// Mock client that fails a specified number of times with 500 error before succeeding
private final class RetryTestClient: HTTPClient, @unchecked Sendable {
  private let failuresBeforeSuccess: Int
  private(set) var attemptCount = 0
  private let lock = NSLock()

  init(failuresBeforeSuccess: Int) {
    self.failuresBeforeSuccess = failuresBeforeSuccess
  }

  func execute(_ request: HTTPRequest) async throws -> HTTPResponse {
    lock.lock()
    attemptCount += 1
    let currentAttempt = attemptCount
    lock.unlock()

    if currentAttempt <= failuresBeforeSuccess {
      throw HTTPError(
        category: .http(.internalServerError),
        request: request
      )
    }

    let userData = #"{"id": 1, "name": "Test User"}"#.data(using: .utf8)!
    return HTTPResponse(
      request: request,
      status: .ok,
      headers: ["Content-Type": "application/json"],
      body: userData
    )
  }
}
```
