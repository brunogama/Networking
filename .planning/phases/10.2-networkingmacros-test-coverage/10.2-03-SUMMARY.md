## Test Patterns Established

### MacroTesting Framework Usage

```swift
final class CacheableMacroTests: XCTestCase {
  override func invokeTest() {
    withMacroTesting(macros: [CacheableMacro.self]) {
      super.invokeTest()
    }
  }

  func testBasicExpansion() {
    assertMacro {
      """
      @Cacheable
      protocol UserAPI { ... }
      """
    } expansion: {
      """
      protocol UserAPI { ... }

      extension UserAPI { ... }
      """
    }
  }
}
```

### Diagnostic Testing Pattern

```swift
func testRequiresProtocol() {
  assertMacro {
    """
    @Cacheable
    struct UserService { ... }
    """
  } diagnostics: {
    """
    @Cacheable
    ┬─────────
    ╰─ 🛑 @Cacheable can only be applied to protocols
    struct UserService { ... }
    """
  }
}
```
