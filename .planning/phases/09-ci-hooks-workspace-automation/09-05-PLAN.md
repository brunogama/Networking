Create `.github/workflows/docs.yml`:

```yaml
name: Documentation

on:
  push:
    branches: [main]
    paths:
      - 'Packages/**/Sources/**/*.swift'
      - 'Documentation.docc/**'
  workflow_dispatch:

concurrency:
  group: docs-${{ github.ref }}
  cancel-in-progress: true

# Sets permissions of the GITHUB_TOKEN to allow deployment to GitHub Pages
permissions:
  contents: read
  pages: write
  id-token: write

jobs:
  build-docs:
    name: Build Documentation
    runs-on: macos-14
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Swift
        uses: swift-actions/setup-swift@v2
        with:
          swift-version: '6.0'

      - name: Build documentation for all packages
        run: |
          mkdir -p docs

          # Build docs for each package that has public API
          for pkg in Networking NetworkingMacros NetworkingWebSocket NetworkingGraphQL; do
            echo "Building docs for $pkg..."
            swift package --package-path "Packages/$pkg" \
              generate-documentation \
              --target "$pkg" \
              --output-path "docs/$pkg" \
              --hosting-base-path "Networking/documentation/$pkg" \
              --transform-for-static-hosting \
              2>/dev/null || echo "Warning: Failed to build docs for $pkg"
          done

          # List generated docs
          echo "## Documentation Build Summary" >> $GITHUB_STEP_SUMMARY
          ls -la docs/ >> $GITHUB_STEP_SUMMARY

      - name: Create index page
        run: |
          cat > docs/index.html << 'EOF'
          <!DOCTYPE html>
          <html>
          <head>
            <meta charset="utf-8">
            <title>ModernNetworking Documentation</title>
            <style>
              body { font-family: -apple-system, BlinkMacSystemFont, sans-serif; max-width: 800px; margin: 50px auto; padding: 20px; }
              h1 { color: #333; }
              .packages { display: grid; grid-template-columns: repeat(auto-fit, minmax(200px, 1fr)); gap: 20px; margin-top: 30px; }
              .package { background: #f5f5f7; padding: 20px; border-radius: 12px; text-decoration: none; color: #333; }
              .package:hover { background: #e5e5e7; }
              .package h3 { margin: 0 0 10px 0; color: #0066cc; }
              .package p { margin: 0; font-size: 14px; color: #666; }
            </style>
          </head>
          <body>
            <h1>ModernNetworking Documentation</h1>
            <p>A production-grade Swift networking library featuring async/await, interceptors, macros, and multi-transport support.</p>
            <div class="packages">
              <a href="Networking/documentation/networking/" class="package">
                <h3>Networking</h3>
                <p>Core async/await HTTP client with middleware and interceptors</p>
              </a>
              <a href="NetworkingMacros/documentation/networkingmacros/" class="package">
                <h3>NetworkingMacros</h3>
                <p>Swift compiler plugin for HTTP client macros</p>
              </a>
              <a href="NetworkingWebSocket/documentation/networkingwebsocket/" class="package">
                <h3>NetworkingWebSocket</h3>
                <p>Actor-based WebSocket client</p>
              </a>
              <a href="NetworkingGraphQL/documentation/networkinggraphql/" class="package">
                <h3>NetworkingGraphQL</h3>
                <p>Type-safe GraphQL client</p>
              </a>
            </div>
          </body>
          </html>
          EOF

      - name: Upload Pages artifact
        uses: actions/upload-pages-artifact@v3
        with:
          path: docs

  deploy-docs:
    name: Deploy to GitHub Pages
    needs: build-docs
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4

  generate-stubs:
    name: Generate Doc Stubs
    runs-on: macos-14
    if: github.event_name == 'push'
    permissions:
      contents: write
      pull-requests: write
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          fetch-depth: 2

      - name: Setup Swift
        uses: swift-actions/setup-swift@v2
        with:
          swift-version: '6.0'

      - name: Install jq
        run: brew install jq

      - name: Detect new types
        id: detect
        run: |
          # Get changed Swift files
          changed=$(git diff --name-only HEAD~1 HEAD -- 'Packages/**/Sources/**/*.swift' || true)

          if [ -z "$changed" ]; then
            echo "has_new_types=false" >> $GITHUB_OUTPUT
            exit 0
          fi

          # Build symbol graph for changed packages
          mkdir -p .build/symbol-graphs
          new_types=""

          for file in $changed; do
            if [[ "$file" =~ ^Packages/([^/]+)/ ]]; then
              pkg="${BASH_REMATCH[1]}"
              swift build --package-path "Packages/$pkg" \
                -Xswiftc -emit-symbol-graph \
                -Xswiftc -emit-symbol-graph-dir -Xswiftc ".build/symbol-graphs/$pkg" 2>/dev/null || true

              # Find new public types
              if [ -f ".build/symbol-graphs/$pkg/$pkg.symbols.json" ]; then
                types=$(jq -r '.symbols[]? | select(.accessLevel == "public" or .accessLevel == "open") | select(.kind.identifier | test("struct|protocol|enum|class")) | .names.title' ".build/symbol-graphs/$pkg/$pkg.symbols.json" 2>/dev/null || true)
                new_types="$new_types $types"
              fi
            fi
          done

          if [ -n "$new_types" ]; then
            echo "has_new_types=true" >> $GITHUB_OUTPUT
            echo "types=$new_types" >> $GITHUB_OUTPUT
          else
            echo "has_new_types=false" >> $GITHUB_OUTPUT
          fi

      - name: Generate stub files
        if: steps.detect.outputs.has_new_types == 'true'
        run: |
          chmod +x scripts/generate-doc-stubs.sh
          ./scripts/generate-doc-stubs.sh ${{ steps.detect.outputs.types }}

      - name: Create PR for stubs
        if: steps.detect.outputs.has_new_types == 'true'
        uses: peter-evans/create-pull-request@v6
        with:
          token: ${{ secrets.GITHUB_TOKEN }}
          commit-message: "docs: add extension stubs for new types"
          title: "docs: Add documentation stubs for new public types"
          body: |
            This PR adds Documentation.docc extension file stubs for newly detected public types.

            **New types detected:**
            ${{ steps.detect.outputs.types }}

            Please review and enhance the stub documentation before merging.
          branch: docs/auto-stubs
          delete-branch: true
          labels: documentation,auto-generated
```
  </action>

```bash
#!/bin/bash
# Generate Documentation.docc extension stubs for new types
# Usage: ./scripts/generate-doc-stubs.sh TypeName1 TypeName2 ...

set -e

DOCS_DIR="Documentation.docc/Extensions"

if [ $# -eq 0 ]; then
  echo "Usage: $0 TypeName1 TypeName2 ..."
  echo "No types provided, nothing to generate."
  exit 0
fi

# Create Extensions directory if needed
mkdir -p "$DOCS_DIR"

created=0
skipped=0

for type in "$@"; do
  # Skip empty or whitespace-only
  type=$(echo "$type" | xargs)
  [ -z "$type" ] && continue

  stub_file="$DOCS_DIR/$type.md"

  if [ -f "$stub_file" ]; then
    echo "skip: $stub_file already exists"
    ((skipped++))
    continue
  fi

  cat > "$stub_file" << EOF
# \`\`$type\`\`

@Metadata {
  @DocumentationExtension(mergeBehavior: append)
}

## Overview

<!-- TODO: Add description of $type -->

## Topics

### Creating $type

<!-- TODO: Add initializer links -->

### Instance Methods

<!-- TODO: Add method links -->

### Instance Properties

<!-- TODO: Add property links -->
EOF

  echo "created: $stub_file"
  ((created++))
done

echo ""
echo "Summary: $created created, $skipped skipped"
```

Make executable: `chmod +x scripts/generate-doc-stubs.sh`
