actor ProgressTrackingMiddleware {

    func scheduleCleanup(for transferId: String) {
        // Cancel any existing cleanup
        cleanupTasks[transferId]?.cancel()

        // Schedule new cleanup with stored reference
        cleanupTasks[transferId] = Task {
            try? await Task.sleep(nanoseconds: 5_000_000_000) // 5 seconds
            guard !Task.isCancelled else { return }
            await self.cleanupStream(transferId)
        }
    }

    func cancelCleanup(for transferId: String) {
        cleanupTasks[transferId]?.cancel()
        cleanupTasks.removeValue(forKey: transferId)

}
