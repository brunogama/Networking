import MacroTemplateKit
import SwiftSyntax
import SwiftSyntaxMacros
import SwiftDiagnostics

/// Macro implementation for @POST HTTP method.
///
/// Generates code for POST requests with request body support.
///
/// Example:
/// ```swift
/// @POST("/users")
/// @Body("user")
/// func createUser(user: User) async throws -> User
/// ```
public struct POSTMacro: PeerMacro, HTTPMacroExpansion {
  public static let config = HTTPMethodConfig.post

  public static func expansion(
    of node: AttributeSyntax,
    providingPeersOf declaration: some DeclSyntaxProtocol,
    in context: some MacroExpansionContext
  ) throws -> [DeclSyntax] {
    try sharedExpansion(of: node, providingPeersOf: declaration, in: context)
  }
}
```

Key changes:
1. Add `HTTPMacroExpansion` conformance
2. Add `static let config = HTTPMethodConfig.post`
3. Replace expansion body with `try sharedExpansion(...)` call
4. Remove all private helper methods (extractPath, extractBodyParameter, extractHeaders, etc.)
5. Keep existing DocC comments
6. Note: Body handling is in HTTPMacroExpansion via config.requiresBody
