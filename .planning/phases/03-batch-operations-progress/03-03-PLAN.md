```swift
  let results = await client.executeBatch(requests)

  // Verify results match original order (not completion order)
  #expect(results.count == 10)
  for (index, result) in results.enumerated() {
    #expect(result.index == index)
    #expect(result.request.url == requests[index].url)
  }
}
```

```swift
@Test func progressTracking_streamsDownloadProgress() async throws {
  let streamManager = ProgressTracking.ProgressStreamManager()
  let transferId = UUID()

  let stream = await streamManager.createProgressStream(
    for: transferId,
    totalBytes: 1000
  )

  var updates: [ProgressTracking.ProgressUpdate] = []
  let collectTask = Task {
    for try await update in stream {
      updates.append(update)
      if update.phase == .completed { break }
    }
  }

  // Simulate progress updates
  try await streamManager.updateProgress(for: transferId, transferredBytes: 250, phase: .downloading)
  try await streamManager.updateProgress(for: transferId, transferredBytes: 500, phase: .downloading)
  try await streamManager.updateProgress(for: transferId, transferredBytes: 1000, phase: .downloading)
  await streamManager.completeProgress(for: transferId, phase: .completed)

  await collectTask.value

  #expect(updates.count >= 4)  // Initial + 3 updates + completed
  #expect(updates.last?.phase == .completed)
}
```

```swift
@Test func batch_withProgressTracking_aggregatesProgress() async throws {
  let client = MockNetworkClient()
  let batchTracker = ProgressTracking.BatchProgressTracker()

  let requests = (1...5).map {
    HTTPRequest(method: .get, url: URL(string: "https://api.example.com/\($0)")!)
  }

  // Start batch tracking
  let transferIds = requests.map { _ in (transferId: UUID(), totalBytes: Int64(1000)) }
  let streams = await batchTracker.startBatchTracking(transferIds)

  // Execute batch with progress callbacks
  var aggregateProgress: [ProgressTracking.ProgressUpdate] = []
  for (index, request) in requests.enumerated() {
    client.stub(request: request, response: HTTPResponse(status: .ok), delay: 0.1)

    // Simulate progress updates
    Task {
      try await Task.sleep(nanoseconds: 50_000_000)  // 50ms
      try await batchTracker.updateBatchProgress(
        transferId: transferIds[index].transferId,
        transferredBytes: 500,
        phase: .downloading
      )
    }
  }

  let results = await client.executeBatch(requests)

  // Verify aggregate progress calculated
  if let aggregate = await batchTracker.getAggregateProgress() {
    #expect(aggregate.totalBytes == 5000)  // 5 × 1000 bytes
    aggregateProgress.append(aggregate)
  }

  #expect(results.count == 5)
  #expect(aggregateProgress.count > 0)
}
```
