  <action>
    Replace the root Package.swift with a workspace manifest that references all child packages:

    **Create new Package.swift**:
    ```swift
    // swift-tools-version: 6.0
    // Workspace manifest for ModernNetworking monorepo

    import PackageDescription

    let package = Package(
      name: "ModernNetworking",
      platforms: [
        .iOS(.v16),
        .macOS(.v13),
        .tvOS(.v16),
        .watchOS(.v9),
      ],
      products: [
        // Re-export all package libraries for convenience
        .library(
          name: "Networking",
          targets: ["NetworkingProxy"]
        ),
        .library(
          name: "NetworkingWebSocket",
          targets: ["NetworkingWebSocketProxy"]
        ),
        .library(
          name: "NetworkingGraphQL",
          targets: ["NetworkingGraphQLProxy"]
        ),
      ],
      dependencies: [
        // Local packages
        .package(path: "Packages/Networking"),
        .package(path: "Packages/NetworkingWebSocket"),
        .package(path: "Packages/NetworkingGraphQL"),
      ],
      targets: [
        // Proxy targets that re-export the actual packages
        .target(
          name: "NetworkingProxy",
          dependencies: [
            .product(name: "Networking", package: "Networking"),
          ],
          path: "Sources/NetworkingProxy"
        ),
        .target(
          name: "NetworkingWebSocketProxy",
          dependencies: [
            .product(name: "NetworkingWebSocket", package: "NetworkingWebSocket"),
          ],
          path: "Sources/NetworkingWebSocketProxy"
        ),
        .target(
          name: "NetworkingGraphQLProxy",
          dependencies: [
            .product(name: "NetworkingGraphQL", package: "NetworkingGraphQL"),
          ],
          path: "Sources/NetworkingGraphQLProxy"
        ),
      ]
    )
    ```

    **Create proxy source files** (required for SPM to build):

    1. Create `Sources/NetworkingProxy/Exports.swift`:
       ```swift
       // Re-export Networking module
       @_exported import Networking
       ```

    2. Create `Sources/NetworkingWebSocketProxy/Exports.swift`:
       ```swift
       // Re-export NetworkingWebSocket module
       @_exported import NetworkingWebSocket
       ```

    3. Create `Sources/NetworkingGraphQLProxy/Exports.swift`:
       ```swift
       // Re-export NetworkingGraphQL module
       @_exported import NetworkingGraphQL
       ```

    **ALTERNATIVE (simpler approach)**: If proxy targets cause issues, use a minimal workspace manifest:
    ```swift
    // swift-tools-version: 6.0
    // Workspace manifest for ModernNetworking monorepo
    // Each package should be consumed directly from Packages/ subdirectories

    import PackageDescription

    let package = Package(
      name: "ModernNetworking",
      platforms: [
        .iOS(.v16),
        .macOS(.v13),
        .tvOS(.v16),
        .watchOS(.v9),
      ],
      products: [],
      dependencies: [
        // Local packages for workspace development
        .package(path: "Packages/Networking"),
        .package(path: "Packages/NetworkingWebSocket"),
        .package(path: "Packages/NetworkingGraphQL"),
      ],
      targets: []
    )
    ```

    The simpler approach means consumers add packages directly:
    ```swift
    .package(url: "https://github.com/brunogama/Networking.git", from: "X.Y.Z")
    // Then in their target:
    .product(name: "Networking", package: "Networking")
    ```
